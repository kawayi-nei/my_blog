```python
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.project import Project


def _other_locale(locale: str) -> str:
    return "en" if locale == "zh" else "zh"


def localize_project(
    project: Project, locale: str, *, detail: bool = False
) -> dict:
    other = _other_locale(locale)

    def pick(field: str) -> str:
        primary = getattr(project, f"{field}_{locale}") or ""
        fallback = getattr(project, f"{field}_{other}") or ""
        return primary or fallback

    data = {
        "slug": project.slug,
        "title": getattr(project, f"title_{locale}"),
        "summary": pick("summary"),
        "cover_url": project.cover_url,
        "tech_stack": project.tech_stack,
        "featured": project.featured,
        "started_on": project.started_on,
        "ended_on": project.ended_on,
    }
    if detail:
        data.update(
            {
                "content_md": pick("content_md"),
                "repo_url": project.repo_url,
                "demo_url": project.demo_url,
            }
        )
    return data


async def list_public_projects(
    db: AsyncSession, *, locale: str, featured: bool | None = None
) -> list[Project]:
    filters = [Project.status == "published"]
    if featured is True:
        filters.append(Project.featured.is_(True))

    result = await db.execute(
        select(Project)
        .where(*filters)
        .order_by(
            Project.featured.desc(),
            Project.sort_order.asc(),
            Project.started_on.desc().nullslast(),
            Project.id.desc(),
        )
    )
    return list(result.scalars().all())


async def get_public_project(
    db: AsyncSession, slug: str
) -> Project | None:
    result = await db.execute(
        select(Project).where(
            Project.slug == slug, Project.status == "published"
        )
    )
    return result.scalar_one_or_none()
```

```python
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.ext.asyncio import AsyncSession

from app.db.session import get_db
from app.schemas.project import ProjectDetail, ProjectListItem
from app.services.projects import (
    get_public_project,
    list_public_projects,
    localize_project,
)

router = APIRouter(prefix="/projects", tags=["projects"])


@router.get("", response_model=list[ProjectListItem])
async def list_projects(
    featured: bool | None = None,
    locale: str = Query("zh", pattern="^(zh|en)$"),
    db: AsyncSession = Depends(get_db),
) -> list[ProjectListItem]:
    projects = await list_public_projects(
        db, locale=locale, featured=featured
    )
    return [
        ProjectListItem(**localize_project(project, locale))
        for project in projects
    ]


@router.get("/{slug}", response_model=ProjectDetail)
async def get_project(
    slug: str,
    locale: str = Query("zh", pattern="^(zh|en)$"),
    db: AsyncSession = Depends(get_db),
) -> ProjectDetail:
    project = await get_public_project(db, slug)
    if project is None:
        raise HTTPException(status_code=404, detail="项目不存在")
    return ProjectDetail(
        **localize_project(project, locale, detail=True)
    )
```

```python
SLUG_PATTERN = r"^[a-z0-9-]+$"


def _normalize_tech_stack(value: list[str]) -> list[str]:
    normalized: list[str] = []
    for item in value:
        item = item.strip()
        if item and item not in normalized:
            normalized.append(item)
    if len(normalized) > 20:
        raise ValueError("技术栈最多 20 项")
    if any(len(item) > 50 for item in normalized):
        raise ValueError("技术栈每项最多 50 字符")
    return normalized

    slug: str = Field(min_length=1, max_length=200, pattern=SLUG_PATTERN)
    title_zh: str = Field(min_length=1, max_length=200)
    title_en: str = Field(min_length=1, max_length=200)

    summary_en: str = Field(default="", max_length=500)
```

```python
from datetime import datetime, timezone

from fastapi import HTTPException
from sqlalchemy import func, or_, select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import defer

from app.models.project import Project
from app.schemas.admin_project import (
    AdminProjectCreate,
    AdminProjectUpdate,
)


async def list_admin_projects(
    db: AsyncSession,
    *,
    page: int,
    page_size: int,
    status: str | None = None,
    q: str | None = None,
) -> tuple[list[Project], int]:
    filters = []
    if status:
        filters.append(Project.status == status)
    if q:
        pattern = f"%{q}%"
        filters.append(
            or_(
                Project.slug.ilike(pattern),
                Project.title_zh.ilike(pattern),
                Project.title_en.ilike(pattern),
            )
        )

    total = (
        await db.scalar(
            select(func.count()).select_from(Project).where(*filters)
        )
        or 0
    )
    result = await db.execute(
        select(Project)
        .where(*filters)
        .options(
            defer(Project.content_md_zh),
            defer(Project.content_md_en),
        )
        .order_by(Project.updated_at.desc(), Project.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
    )
    return list(result.scalars().all()), total


async def get_admin_project(
    db: AsyncSession, project_id: int
) -> Project | None:
    result = await db.execute(
        select(Project).where(Project.id == project_id)
    )
    return result.scalar_one_or_none()


async def _ensure_slug_available(
    db: AsyncSession, slug: str, *, exclude_id: int | None = None
) -> None:
    query = select(Project.id).where(Project.slug == slug)
    if exclude_id is not None:
        query = query.where(Project.id != exclude_id)
    if await db.scalar(query) is not None:
        raise HTTPException(status_code=409, detail="slug 已存在")


def _validate_date_order(
    started_on, ended_on
) -> None:
    if started_on and ended_on and ended_on < started_on:
        raise HTTPException(
            status_code=422, detail="结束日期不能早于开始日期"
        )


async def create_admin_project(
    db: AsyncSession, data: AdminProjectCreate
) -> Project:
    await _ensure_slug_available(db, data.slug)
    project = Project(
        slug=data.slug,
        title_zh=data.title_zh,
        title_en=data.title_en,
        summary_zh=data.summary_zh,
        summary_en=data.summary_en,
        content_md_zh=data.content_md_zh,
        content_md_en=data.content_md_en,
        cover_url=data.cover_url,
        tech_stack=data.tech_stack,
        repo_url=data.repo_url,
        demo_url=data.demo_url,
        started_on=data.started_on,
        ended_on=data.ended_on,
        featured=data.featured,
        sort_order=data.sort_order,
        status=data.status,
    )
    db.add(project)
    try:
        await db.commit()
    except IntegrityError:
        await db.rollback()
        raise HTTPException(status_code=409, detail="slug 已存在")
    await db.refresh(project)
    return project


async def update_admin_project(
    db: AsyncSession, project_id: int, data: AdminProjectUpdate
) -> Project:
    project = await get_admin_project(db, project_id)
    if project is None:
        raise HTTPException(status_code=404, detail="项目不存在")

    changes = data.model_dump(exclude_unset=True)
    nullable = {
        "cover_url",
        "repo_url",
        "demo_url",
        "started_on",
        "ended_on",
    }
    changes = {
        key: value
        for key, value in changes.items()
        if value is not None or key in nullable
    }

    new_slug = changes.get("slug")
    if new_slug is not None and new_slug != project.slug:
        await _ensure_slug_available(
            db, new_slug, exclude_id=project.id
        )

    started_on = changes.get("started_on", project.started_on)
    ended_on = changes.get("ended_on", project.ended_on)
    _validate_date_order(started_on, ended_on)

    for field, value in changes.items():
        setattr(project, field, value)

    project.updated_at = datetime.now(timezone.utc)
    try:
        await db.commit()
    except IntegrityError:
        await db.rollback()
        raise HTTPException(status_code=409, detail="slug 已存在")
    await db.refresh(project)
    return project


async def delete_admin_project(
    db: AsyncSession, project_id: int
) -> None:
    project = await get_admin_project(db, project_id)
    if project is None:
        raise HTTPException(status_code=404, detail="项目不存在")
    await db.delete(project)
    await db.commit()
```

```python
    status: str | None = Query(None, pattern="^(draft|published)$"),

        status=status,
```

```tsx
import {useLocale, useTranslations} from 'next-intl';
import {useRouter} from 'next/navigation';
import {useState} from 'react';

export default function DeleteProjectButton({
  id,
  title
}: {
  id: number;
  title: string;
}) {
  const t = useTranslations('admin.projects');
  const locale = useLocale();
  const router = useRouter();
  const [pending, setPending] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function onDelete() {
    if (!window.confirm(t('confirmDelete', {title}))) return;
    setPending(true);
    setError(null);
    try {
      const response = await fetch(`/api/admin/projects/${id}`, {
        method: 'DELETE'
      });
      if (response.status === 401) {
        window.location.assign(`/${locale}/admin/login`);
        return;
      }
      if (response.ok || response.status === 404) {
        router.refresh();
        return;
      }
      setError(t('deleteFailed'));
    } catch {
      setError(t('deleteFailed'));
    } finally {
      setPending(false);
    }
  }

  return (
    <span className="inline-flex items-center gap-2">
      <button
        type="button"
        onClick={onDelete}
        disabled={pending}
        className="text-red-400 hover:underline disabled:opacity-60"
      >
        {t('delete')}
      </button>
      {error ? (
        <span role="alert" className="text-red-400">
          {error}
        </span>
      ) : null}
    </span>
  );
}
```

```tsx
import {useRef, useState} from 'react';

export default function CoverPicker({
  value,
  onChange
}: {
  value: string | null;
  onChange: (value: string | null) => void;
}) {
  const t = useTranslations('admin.projects.form.cover');
  const locale = useLocale();
  const fileRef = useRef<HTMLInputElement>(null);
  const [open, setOpen] = useState(false);
  const [items, setItems] = useState<MediaItem[] | null>(null);
  const [loading, setLoading] = useState(false);
  const [uploading, setUploading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function upload(event: React.ChangeEvent<HTMLInputElement>) {
    const file = event.target.files?.[0];
    event.target.value = '';
    if (!file) return;

    setUploading(true);
    setError(null);
    try {
      const form = new FormData();
      form.append('file', file);
      const response = await fetch('/api/admin/uploads', {
        method: 'POST',
        body: form
      });
      if (response.status === 401) {
        window.location.assign(`/${locale}/admin/login`);
        return;
      }
      if (!response.ok) {
        setError(t('uploadFailed'));
        return;
      }
      const data = (await response.json()) as {url: string};
      onChange(data.url);
    } catch {
      setError(t('uploadFailed'));
    } finally {
      setUploading(false);
    }
  }

  async function toggleLibrary() {
    if (open) {
      setOpen(false);
      return;
    }

    setOpen(true);
    setError(null);
    if (items !== null) return;

    setLoading(true);
    try {
      const response = await fetch('/api/admin/media?page_size=100');
      if (response.status === 401) {
        window.location.assign(`/${locale}/admin/login`);
        return;
      }
      if (!response.ok) {
        setError(t('loadFailed'));
        return;
      }
      const data = (await response.json()) as {items: MediaItem[]};
      setItems(data.items);
    } catch {
      setError(t('loadFailed'));
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="grid gap-3">
      <span className="font-mono text-sm">{t('label')}</span>

      {value ? (
        <img
          src={value}
          alt={t('preview')}
          className="h-36 w-full max-w-xs rounded border border-border object-cover"
        />
      ) : (
        <p className="text-xs text-muted">{t('noCover')}</p>
      )}

      <div className="flex flex-wrap gap-2 font-mono text-xs">
        <button
          type="button"
          onClick={() => fileRef.current?.click()}
          disabled={uploading}
          className="rounded border border-border px-3 py-1 hover:border-accent disabled:opacity-60"
        >
          {uploading ? t('uploading') : t('upload')}
        </button>

          type="button"
          onClick={toggleLibrary}
          aria-expanded={open}
          className="rounded border border-border px-3 py-1 hover:border-accent"
        >
          {open ? t('close') : t('choose')}
        </button>
        {value ? (
          <button
            type="button"
            onClick={() => onChange(null)}
            className="rounded border border-border px-3 py-1 text-red-400 hover:border-red-400"
          >
            {t('clear')}
          </button>
        ) : null}

          ref={fileRef}
          type="file"
```
